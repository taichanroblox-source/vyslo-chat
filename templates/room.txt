<script>
const roomId = {{ room_id | tojson }};
const box = document.getElementById("messages");
const form = document.getElementById("sendForm");
const input = document.getElementById("content");
const notice = document.getElementById("notice");
const emojis = ["👍","❤️","😂","🎉","😮","👏"];
let me = "";
let roomAdmin = false;
let webAdmin = false;
let lastSignature = "";

async function api(url, options = {}) {
  const response = await fetch(url, {
    ...options,
    headers: {"Content-Type":"application/json", ...(options.headers || {})}
  });
  const data = await response.json().catch(() => ({}));
  if (!response.ok) throw new Error(data.error || "エラーが発生したよ");
  return data;
}
function showError(error){ notice.textContent = error.message || String(error); }

async function loadInfo(){
  const data = await api("/api/room-info?room_id=" + encodeURIComponent(roomId));
  me = data.username;
  roomAdmin = !!data.is_room_admin;
  webAdmin = !!data.is_web_admin;
  document.getElementById("commandPanel").style.display = roomAdmin ? "block" : "none";
  const history = document.getElementById("deletedPanel");
  history.style.display = (roomAdmin || webAdmin) ? "block" : "none";
  if (roomAdmin || webAdmin) await loadDeleted();
}

async function loadMessages(){
  try {
    const data = await api("/api/messages?room_id=" + encodeURIComponent(roomId));
    const messages = data.messages || [];
    const signature = JSON.stringify(messages);
    if (signature === lastSignature) return;
    const wasNearBottom = box.scrollHeight - box.scrollTop - box.clientHeight < 80;
    lastSignature = signature;
    box.replaceChildren();

    for (const message of messages) {
      const row = document.createElement("div");
      row.className = "message";
      const heading = document.createElement("div");
      const author = document.createElement("strong");
      author.textContent = message.username;
      const time = document.createElement("span");
      time.className = "time";
      time.textContent = message.created_at || "";
      heading.append(author,time);
      const content = document.createElement("div");
      content.textContent = message.content;
      row.append(heading,content);

      const reactions = document.createElement("div");
      reactions.className = "reaction-row";
      for (const emoji of emojis) {
        const found = (message.reactions || []).find(r => r.emoji === emoji);
        const button = document.createElement("button");
        button.type = "button";
        button.className = "secondary";
        button.textContent = emoji + (found ? " " + found.count : "");
        button.addEventListener("click", async () => {
          try {
            await api("/api/react", {method:"POST",body:JSON.stringify({
              room_id:roomId,message_id:message.id,emoji
            })});
            lastSignature = "";
            await loadMessages();
          } catch(e){showError(e);}
        });
        reactions.append(button);
      }
      row.append(reactions);

      if (message.username === me || roomAdmin || webAdmin) {
        const actions = document.createElement("div");
        actions.className = "message-actions";
        const del = document.createElement("button");
        del.type = "button";
        del.className = "danger";
        del.textContent = "削除";
        del.addEventListener("click", async () => {
          try {
            await api("/api/delete-message", {method:"POST",body:JSON.stringify({
              room_id:roomId,message_id:message.id
            })});
            lastSignature = "";
            await loadMessages();
            if (roomAdmin || webAdmin) await loadDeleted();
          } catch(e){showError(e);}
        });
        actions.append(del);
        row.append(actions);
      }
      box.append(row);
    }
    if (wasNearBottom) box.scrollTop = box.scrollHeight;
  } catch(error){showError(error);}
}

async function loadMembers(){
  try{
    const data = await api("/api/members?room_id=" + encodeURIComponent(roomId));
    const list = document.getElementById("members");
    list.replaceChildren();
    for(const member of (data.members || [])){
      const li = document.createElement("li");
      li.textContent = member.username + (member.role === "admin" ? " ★ 管理者" : "");
      if(member.role === "admin") li.className = "member-admin";
      list.append(li);
    }
  }catch(e){showError(e);}
}

async function loadHands(){
  try{
    const data = await api("/api/hand?room_id=" + encodeURIComponent(roomId));
    const list = document.getElementById("hands");
    list.replaceChildren();
    for(const hand of (data.hands || [])){
      const li = document.createElement("li");
      li.textContent = hand.username;
      list.append(li);
    }
  }catch(e){showError(e);}
}

async function loadDeleted(){
  try{
    const data = await api("/api/deleted-messages?room_id=" + encodeURIComponent(roomId));
    const list = document.getElementById("deletedMessages");
    list.replaceChildren();
    if(!(data.messages || []).length){
      list.textContent = "履歴はまだないよ";
      return;
    }
    for(const message of data.messages){
      const item = document.createElement("div");
      item.className = "deleted-item";
      const title = document.createElement("strong");
      title.textContent = message.username + "：";
      const content = document.createElement("span");
      content.textContent = message.content;
      const meta = document.createElement("div");
      meta.className = "muted";
      meta.textContent = "削除者：" + message.deleted_by + " / 削除日時：" + message.deleted_at;
      item.append(title,content,meta);
      list.append(item);
    }
  }catch(e){showError(e);}
}


// 浮遊絵文字パネル
const floatingLayer = document.getElementById("floatingEmojiLayer");
const emojiPanel = document.getElementById("emojiPanel");
const emojiToggle = document.getElementById("emojiToggle");
const emojiStatus = document.getElementById("emojiStatus");
const floatingEmojiChoices = ["👍","❤️","😂","🎉","😮","👏","🔥","✨","💯","🥳"];
const seenFloatingEmojiIds = new Set();
let myFloatingEmojiActive = false;

emojiToggle.addEventListener("click", () => {
  emojiPanel.hidden = !emojiPanel.hidden;
  emojiToggle.setAttribute("aria-expanded", String(!emojiPanel.hidden));
});

for (const emoji of floatingEmojiChoices) {
  const button = document.createElement("button");
  button.type = "button";
  button.textContent = emoji;
  button.title = emoji + " を浮かべる";
  button.addEventListener("click", async () => {
    if (myFloatingEmojiActive) return;
    button.disabled = true;
    try {
      await api("/api/floating-emojis", {
        method: "POST",
        body: JSON.stringify({room_id: roomId, emoji})
      });
      emojiPanel.hidden = true;
      emojiToggle.setAttribute("aria-expanded", "false");
      notice.textContent = "";
      await pollFloatingEmojis();
    } catch (e) {
      showError(e);
    } finally {
      await pollFloatingEmojis();
    }
  });
  emojiPanel.append(button);
}

async function pollFloatingEmojis() {
  try {
    const rows = await api(
      "/api/floating-emojis?room_id=" + encodeURIComponent(roomId)
    );
    const activeRows = Array.isArray(rows) ? rows : [];
    myFloatingEmojiActive = activeRows.some(item => item.username === me);

    emojiStatus.textContent = myFloatingEmojiActive ? "表示中…" : "絵文字";
    emojiToggle.disabled = myFloatingEmojiActive;
    for (const button of emojiPanel.querySelectorAll("button")) {
      button.disabled = myFloatingEmojiActive;
    }

    for (const item of activeRows) {
      if (seenFloatingEmojiIds.has(item.id)) continue;
      seenFloatingEmojiIds.add(item.id);

      const float = document.createElement("div");
      float.className = "floating-emoji";
      float.style.left = (5 + Math.random() * 85) + "%";

      const face = document.createElement("span");
      face.className = "emoji-face";
      face.textContent = item.emoji;

      const name = document.createElement("span");
      name.className = "emoji-name";
      name.textContent = item.username;

      float.append(face, name);
      floatingLayer.append(float);
      float.addEventListener("animationend", () => float.remove(), {once:true});
      setTimeout(() => float.remove(), 4500);
    }

    // ID記録が大きくなりすぎないようにする
    if (seenFloatingEmojiIds.size > 500) seenFloatingEmojiIds.clear();
  } catch (e) {
    // 一時的な通信エラーでは文章チャットを止めない
  }
}

form.addEventListener("submit", async event => {
  event.preventDefault();
  const content = input.value.trim();
  if (!content) return;
  document.getElementById("sendButton").disabled = true;
  try {
    await api("/api/messages", {method:"POST",body:JSON.stringify({room_id:roomId,content})});
    input.value = "";
    notice.textContent = "";
    lastSignature = "";
    await loadMessages();
  } catch(error){showError(error);}
  finally {document.getElementById("sendButton").disabled = false;input.focus();}
});

document.getElementById("handButton").addEventListener("click", async () => {
  try{
    const result = await api("/api/hand",{method:"POST",body:JSON.stringify({room_id:roomId})});
    document.getElementById("handButton").textContent = result.raised ? "✋ 挙手を取り消す" : "✋ 挙手する";
    await loadHands();
  }catch(e){showError(e);}
});

document.getElementById("commandForm").addEventListener("submit", async event => {
  event.preventDefault();
  const command = document.getElementById("commandInput").value.trim();
  if(!command) return;
  if(!confirm("このコマンドを実行する？\n" + command)) return;
  try{
    const result = await api("/api/room-command",{method:"POST",body:JSON.stringify({room_id:roomId,command})});
    notice.textContent = result.message || "実行したよ";
    document.getElementById("commandInput").value = "";
    lastSignature = "";
    await loadMessages();
    await loadMembers();
    await loadDeleted();
  }catch(e){showError(e);}
});

(async()=>{
  try{await loadInfo();await loadMessages();await loadMembers();await loadHands();await pollFloatingEmojis();}
  catch(e){showError(e);}
})();
setInterval(()=>{loadMessages();loadMembers();loadHands();},3000);
setInterval(pollFloatingEmojis, 500);
</script>
